@startuml

skinparam classAttributeIconSize 0

interface CriterioBusqueda {
    + cumple(item : Item) : boolean
}

class Catalogo {
    - items : List<Item>
    + agregarItem(item : Item)
    + buscar(criterio : CriterioBusqueda) : List<Item>
}

class PorNombre {
    - textoBuscado : String
}

class PorPrecioMaximo {
    - precioMaximo : double
}

class PorCategoria {
    - categoria : Categoria
}

class PorDisponibilidad {
    - consultaInventario : ConsultaDisponibilidadInventario
}

class ConsultaDisponibilidadInventario {
    - inventario : Inventario
    + tieneStock(sku : String) : boolean
}

class Inventario {
    - stockDeInventario : Map<String,Integer>
    + incrementarStock(articulos : Map<String,Integer>)
    + decrementarStock(articulos : Map<String,Integer>)
    + tieneStock(sku : String) : boolean
}

abstract class Item {
    + getNombre() : String
    + coincideNombre(textoBuscado : String) : boolean
    + getPrecioBaseCalculado() : double
    + getCategoria() : Categoria
    + getResumenDeSku() : Map<String,Integer>
}

class Producto {
    - sku : String
    - categoria : Categoria
    + coincideNombre(textoBuscado : String) : boolean
}

class Paquete {
    - items : List<Item>
    + coincideNombre(textoBuscado : String) : boolean
}

enum Categoria {
    ELECTRONICA
    HOGAR
    JUGUETES
    ROPA
    DEPORTES
    LIBROS
    OTROS
}

class And {
    - criterio1 : CriterioBusqueda
    - criterio2 : CriterioBusqueda
}

class Or {
    - criterio1 : CriterioBusqueda
    - criterio2 : CriterioBusqueda
}

class Not {
    - criterio : CriterioBusqueda
}

CriterioBusqueda <|.. PorNombre
CriterioBusqueda <|.. PorPrecioMaximo
CriterioBusqueda <|.. PorCategoria
CriterioBusqueda <|.. PorDisponibilidad

CriterioBusqueda <|.. And
CriterioBusqueda <|.. Or
CriterioBusqueda <|.. Not

Catalogo o-- Item
Catalogo ..> CriterioBusqueda

And --> CriterioBusqueda
Or --> CriterioBusqueda
Not --> CriterioBusqueda

PorDisponibilidad --> ConsultaDisponibilidadInventario
ConsultaDisponibilidadInventario --> Inventario

Item <|-- Producto
Item <|-- Paquete

Paquete o-- Item

Producto --> Categoria
PorCategoria --> Categoria

@enduml
